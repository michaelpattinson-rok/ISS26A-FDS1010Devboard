cmake_minimum_required(VERSION 3.20)
set(PICO_BOARD pico CACHE STRING "pico, pico_w, pico2 or pico2_w")
set(FDS_SUPPORTED_BOARDS pico pico_w pico2 pico2_w)
if(NOT PICO_BOARD IN_LIST FDS_SUPPORTED_BOARDS)
    message(FATAL_ERROR "Select pico, pico_w, pico2 or pico2_w")
endif()
if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
    message(FATAL_ERROR "Set PICO_SDK_PATH to an EXISTING Pico SDK 2.x; downloads disabled")
endif()
set(PICO_SDK_FETCH_FROM_GIT OFF CACHE BOOL "Never fetch SDK" FORCE)
set(PICO_TINYUSB_FETCH_FROM_GIT OFF CACHE BOOL "Never fetch TinyUSB" FORCE)
set(PICO_NO_PICOTOOL 1 CACHE BOOL "Do not download/build picotool; ELF/bin/hex outputs only" FORCE)
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(fds1010_reve C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
if(NOT PICO_PLATFORM MATCHES "^(rp2040|rp2350.*)$")
    message(FATAL_ERROR "RP2040 or RP2350 target required")
endif()
option(FDS_ENABLE_FIXTURE "Enable manual GP15 fixture mode; detector must be disconnected" OFF)
add_executable(fds1010_reve main.c pulse_estimator.c)
target_compile_options(fds1010_reve PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(fds1010_reve PRIVATE FDS_ENABLE_FIXTURE=$<BOOL:${FDS_ENABLE_FIXTURE}>)
target_link_libraries(fds1010_reve pico_stdlib hardware_adc hardware_dma hardware_irq hardware_clocks m)
# No CYW43/network library: radio is not initialized; no direct GPIO23 writes.
pico_enable_stdio_usb(fds1010_reve 1)
pico_enable_stdio_uart(fds1010_reve 0)
pico_add_extra_outputs(fds1010_reve)
